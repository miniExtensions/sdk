export function nativeIdentityProtocolMetadata(type: PostgresType) {
    const array: boolean | undefined = type.nativeArrayProtocol;
    const enumProtocol: boolean | undefined = type.nativeEnumProtocol;
    const attested: PostgresType = {
        ...type,
        nativeArrayProtocol: true,
        nativeEnumProtocol: false,
    };
    const invalid: PostgresType = {
        ...type,
        // @ts-expect-error Native protocol metadata cannot be inferred from a caller's type-name string.
        nativeEnumProtocol: 'enum_recv',
    };
    return { array, enumProtocol, attested, invalid };
}

export function nativeArrayIdentityTransport(
    binary: NonNullable<PostgresRecordIdentity['values'][number]['binary']>
) {
    const format: 'postgres-array-binary-v1' = binary.format;
    const hex: string = binary.hex;
    const sourceConnection: string = binary.source.connectionId;
    const identity: PostgresRecordIdentity = {
        formatVersion: 1,
        type: 'primary-key',
        values: [{ columnName: 'key', value: '{NULL}', binary }],
    };
    const invalid = {
        ...binary,
        // @ts-expect-error Binary identity input must remain portable JSON hex.
        hex: new Uint8Array(12),
    } satisfies typeof binary;
    return { format, hex, sourceConnection, identity, invalid };
}

export function inspectNativeTextPresentation(
    presentation: PostgresTextColumnPresentation
) {
    const stored: 'text' = presentation.storedValue;
    const mutation: 'excluded' = presentation.attachmentMutation;
    const output: 'excluded' = presentation.generatedAttachmentOutput;
    if (presentation.renderer === 'public-url-attachments') {
        const input: 'read-only' = presentation.nativeValueInput;
        const opening: 'existing-public-http-urls' = presentation.fileOpening;
        return { stored, mutation, output, input, opening };
    }
    const input: 'applicable' = presentation.nativeValueInput;
    return { stored, mutation, output, input };
}

export function inspectPrivateDraft(draft: PostgresNativeDraft) {
    const nativeSource: 'postgres' = draft.source.type;
    const currentVersion: 106 = draft.version;
    if (draft.type === 'form') {
        const fields: PostgresConfiguredColumn[] = draft.state.formFields;
        const generation: number = draft.relationOid;
        // @ts-expect-error Form state does not acquire Portal tabs.
        draft.state.tabs;
        return { nativeSource, currentVersion, fields, generation };
    }
    const tabs: PostgresPortalTab[] = draft.state.tabs;
    // @ts-expect-error Portal state does not acquire Form fields.
    draft.state.formFields;
    return { nativeSource, currentVersion, tabs };
}
